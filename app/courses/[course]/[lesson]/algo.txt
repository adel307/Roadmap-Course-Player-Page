getLessons(Course.id):
    // try store first
    lessons = store.LessonsByCourse(Course.id)   // lessonsByCourse
    if (lessons && lessons.length > 0): return lessons

    // fallback to server
    lessons = fetchLessonsForCourse(
            Course.id,
            Course.playlistUrl,
            async (cId, pId) => {
                const res = await fetchLessons(cId, pId);
                return res.lessons;
            },
        )
    if (!lessons): return null
    store.addLessons(lessons)
    return lessons
